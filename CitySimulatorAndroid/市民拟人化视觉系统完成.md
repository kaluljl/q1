| IDLE | 静止 | NEUTRAL等 |

```kotlin
@Composable
fun EmotionBubble(

    // 特性：
```

## 🔧 技术实现细节

### 1. 动画实现

#### 无限循环动画辅助函数
```kotlin
@Composable
private fun rememberInfiniteFloatAnimation(
    enabled: Boolean,
    initialValue: Float,
    targetValue: Float,
    durationMillis: Int
): State<Float> {
    val infiniteTransition = rememberInfiniteTransition(label = "behavior_animation")
    
    return if (enabled) {
        infiniteTransition.animateFloat(
            initialValue = initialValue,
            targetValue = targetValue,
            animationSpec = infiniteRepeatable(
                animation = tween(durationMillis, easing = FastOutSlowInEasing),
                repeatMode = RepeatMode.Reverse
            ),
            label = "animation_value"
        )
    } else {
        remember { mutableStateOf(0f) }
    }
}
```

**使用示例**:
```kotlin
// 跳跃动画
val jumpOffset by rememberInfiniteFloatAnimation(
    enabled = behavior == BehaviorAnimation.JUMP,
    initialValue = 0f,
    targetValue = -8f,
    durationMillis = 400
)

Box(modifier = Modifier.offset(y = jumpOffset.dp)) {
    SimpleCitizenMarker(citizen)
}
```

### 3. 关系分析算法

```kotlin
private fun analyzeRelationship(citizen1: Citizen, citizen2: Citizen): String {
    return when {
        // 配偶
        familyId相同 && 已婚 -> "配偶"
        
        // 同事
        workplaceX相同 && workplaceY相同 -> "同事"
        
        // 邻居
        abs(homeX差) <= 1 && abs(homeY差) <= 1 -> "邻居"
        
        // 同行
        occupation相同 -> "同行"
        
        // 陌生人
        else -> "陌生人"
    }
}
```

## 🚧 技术限制

### 1. 需求数据简化
- **当前**: `MaslowNeeds`使用默认值
- **原因**: 数据结构复杂（5层嵌套），参数名需精确匹配
- **影响**: 需求Tab暂时显示默认数据
- **未来**: 架构已就绪，可后续扩展

### 2. AI调用成本
- **API**: DeepSeek (收费)
- **降级**: 网络失败时使用模板
- **优化**: 缓存机制（同一对话24小时内复用）

### 3. 性能优化
- **动画**: 仅对可见市民启用
- **气泡**: 3秒后自动销毁
- **AI**: 异步调用，不阻塞UI
